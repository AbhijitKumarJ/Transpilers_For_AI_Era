from __future__ import annotations
import ast
from dataclasses import dataclass
import importlib.util
import marshal
from pathlib import Path
import struct
import sys
import time
import types
from psc.codegen.py_ast_builder import PythonAstBuilder
from psc.codegen.topo_sorter import NamespacePackagePlanner, PlannedPythonModule
from psc.semantics.symbols import GlobalSymbolTable


@dataclass(slots=True)
class EmittedArtifact:
    """Represents a compiled Python module and its line-pinned bytecode artifact."""
    namespace: tuple[str, ...]
    module_rel_path: str
    py_path: Path
    pyc_path: Path | None
    pycache_pyc_path: Path | None
    py_source: str
    code_object: types.CodeType
    primary_ps_filename: str


class ArtifactEmitter:
    """
    Pass 3B — Python 3.11+ Package & Line-Pinned PEP 552 `.pyc` Bytecode Emitter:
      1. Emits formatted Python 3.11+ source files (`.py`) grouped by logical namespace,
         automatically creating intermediate `__init__.py` package markers.
      2. Compiles the `.ps`-span-stamped `ast.Module` directly into a Python `CodeType`
         object before unparsing—preserving the exact `.ps` line numbers (`lineno`) and
         pinning `co_filename` on every function/class code object to its originating
         `.ps` source file.
      3. Writes PEP 552 `.pyc` files (both standalone `<module>.pyc` and inside
         `__pycache__/<stem>.cpython-<XY>.pyc`) so CPython runtime stack traces
         natively display `File "Services/OrderCache.ps", line 24` with zero runtime hooks.
    """

    def __init__(self, symtab: GlobalSymbolTable) -> None:
        self.symtab = symtab
        self.ast_builder = PythonAstBuilder(symtab)

    def emit_all(
        self,
        out_dir: Path,
        emit_pyc: bool = True,
    ) -> list[EmittedArtifact]:
        out_dir = out_dir.resolve()
        out_dir.mkdir(parents=True, exist_ok=True)

        planner = NamespacePackagePlanner(self.symtab)
        planned_modules = planner.plan_modules()
        artifacts: list[EmittedArtifact] = []

        for rel_path, planned in sorted(planned_modules.items()):
            artifact = self.emit_module(planned, out_dir=out_dir, emit_pyc=emit_pyc)
            artifacts.append(artifact)

        return artifacts

    def emit_module(
        self,
        planned: PlannedPythonModule,
        out_dir: Path,
        emit_pyc: bool = True,
    ) -> EmittedArtifact:
        module_ast = self.ast_builder.build_module(planned)

        # Build a map of class/method names -> originating `.ps` filename in case
        # multiple `.ps` files contributed types to the same logical namespace.
        qualname_to_ps_file: dict[str, str] = {}
        for type_sym in planned.types:
            ps_file = type_sym.span.ps_filename
            qualname_to_ps_file[type_sym.name] = ps_file

        # 1. Compile the `.ps`-line-stamped `ast.Module` BEFORE `ast.unparse` so
        #    every bytecode instruction retains its exact `.ps` `lineno` and `col_offset`!
        raw_code = compile(
            module_ast,
            filename=planned.primary_ps_filename,
            mode="exec",
            dont_inherit=True,
        )
        pinned_code = self._pin_nested_code_filenames(
            raw_code,
            default_ps_filename=planned.primary_ps_filename,
            qualname_to_ps_file=qualname_to_ps_file,
        )

        # 2. Generate clean, human-readable Python 3.11+ source code
        header = (
            f"# Auto-generated by PythonSharp (psc) — "
            f"Bytecode line-pinned to {planned.primary_ps_filename}\n"
        )
        py_source = header + ast.unparse(module_ast) + "\n"

        py_path = out_dir / planned.module_rel_path
        py_path.parent.mkdir(parents=True, exist_ok=True)
        self._ensure_package_init_files(out_dir, py_path.parent)
        py_path.write_text(py_source, encoding="utf-8")

        pyc_path: Path | None = None
        pycache_pyc_path: Path | None = None

        if emit_pyc:
            stat = py_path.stat()
            mtime = int(stat.st_mtime)
            source_size = stat.st_size & 0xFFFFFFFF
            pyc_bytes = self.build_pep552_pyc_bytes(
                pinned_code, mtime=mtime, source_size=source_size
            )

            # Write standalone `<module>.pyc`
            pyc_path = py_path.with_suffix(".pyc")
            pyc_path.write_bytes(pyc_bytes)

            # Also write PEP 3147 `__pycache__/<stem>.<cache_tag>.pyc` so standard
            # `import <package.module>` loads our `.ps`-pinned bytecode!
            cache_tag = sys.implementation.cache_tag or f"cpython-{sys.version_info.major}{sys.version_info.minor}"
            pycache_dir = py_path.parent / "__pycache__"
            pycache_dir.mkdir(parents=True, exist_ok=True)
            pycache_pyc_path = pycache_dir / f"{py_path.stem}.{cache_tag}.pyc"
            pycache_pyc_path.write_bytes(pyc_bytes)

        return EmittedArtifact(
            namespace=planned.namespace,
            module_rel_path=planned.module_rel_path,
            py_path=py_path,
            pyc_path=pyc_path,
            pycache_pyc_path=pycache_pyc_path,
            py_source=py_source,
            code_object=pinned_code,
            primary_ps_filename=planned.primary_ps_filename,
        )

    @classmethod
    def _pin_nested_code_filenames(
        cls,
        code_obj: types.CodeType,
        default_ps_filename: str,
        qualname_to_ps_file: dict[str, str],
        active_ps_filename: str | None = None,
    ) -> types.CodeType:
        """
        Recursively walks `code_obj.co_consts` and pins `co_filename` on every nested
        class and function `CodeType` object to the exact `.ps` file that declared it.
        """
        target_file = qualname_to_ps_file.get(
            code_obj.co_name, active_ps_filename or default_ps_filename
        )
        new_consts = []
        changed = code_obj.co_filename != target_file

        for const in code_obj.co_consts:
            if isinstance(const, types.CodeType):
                pinned_child = cls._pin_nested_code_filenames(
                    const,
                    default_ps_filename=default_ps_filename,
                    qualname_to_ps_file=qualname_to_ps_file,
                    active_ps_filename=target_file,
                )
                if pinned_child is not const:
                    changed = True
                new_consts.append(pinned_child)
            else:
                new_consts.append(const)

        if changed:
            return code_obj.replace(
                co_filename=target_file,
                co_consts=tuple(new_consts),
            )
        return code_obj

    @staticmethod
    def build_pep552_pyc_bytes(
        code_obj: types.CodeType,
        mtime: int | None = None,
        source_size: int = 0,
    ) -> bytes:
        """
        Constructs a standard 16-byte PEP 552 timestamp-invalidated `.pyc` payload:
          - 4 bytes: `importlib.util.MAGIC_NUMBER`
          - 4 bytes: bitfield (`0` = timestamp-based invalidation)
          - 4 bytes: 32-bit timestamp (`mtime`)
          - 4 bytes: 32-bit source file size (`source_size`)
          - N bytes: `marshal.dumps(code_obj)`
        """
        ts = int(time.time()) if mtime is None else (mtime & 0xFFFFFFFF)
        header = importlib.util.MAGIC_NUMBER + struct.pack(
            "<III", 0, ts, source_size & 0xFFFFFFFF
        )
        return header + marshal.dumps(code_obj)

    @staticmethod
    def load_code_from_pyc(pyc_path: Path) -> types.CodeType:
        """Loads and unmarshals the `CodeType` object from a 16-byte header `.pyc` file."""
        raw = pyc_path.read_bytes()
        code_obj = marshal.loads(raw[16:])
        assert isinstance(code_obj, types.CodeType)
        return code_obj

    @staticmethod
    def _ensure_package_init_files(root_out_dir: Path, target_dir: Path) -> None:
        curr = target_dir.resolve()
        root = root_out_dir.resolve()
        while curr != root and root in curr.parents:
            init_file = curr / "__init__.py"
            if not init_file.exists():
                init_file.write_text("# Auto-generated PythonSharp package\n", encoding="utf-8")
            curr = curr.parent

from __future__ import annotations
import argparse
from pathlib import Path
import sys
import traceback
from typing import Sequence
from psc.codegen import ArtifactEmitter, EmittedArtifact, NamespacePackagePlanner, PythonAstBuilder
from psc.diagnostics import DiagnosticEngine
from psc.frontend import CompilationUnit, Parser
from psc.project import PsProjectConfig, load_project
from psc.semantics import GlobalSymbolTable, analyze_compilation_units


def _find_default_stdlib_dir() -> Path | None:
    """Locates `stdlib/PythonSharp.Core.d.ps` relative to the package or working directory."""
    candidates = [
        Path(__file__).resolve().parent.parent.parent / "stdlib",
        Path.cwd() / "stdlib",
    ]
    for cand in candidates:
        if cand.exists() and any(cand.glob("*.d.ps")):
            return cand
    return None


def _discover_inputs(target: Path, out_dir: Path | None) -> PsProjectConfig:
    stdlib_dir = _find_default_stdlib_dir()
    target = target.resolve()

    if target.is_file() and target.suffix == ".psproj":
        cfg = load_project(target, stdlib_dir=stdlib_dir)
        if out_dir is not None:
            cfg.out_dir = out_dir.resolve()
        return cfg

    if target.is_file() and target.suffix == ".ps":
        stubs = sorted(stdlib_dir.rglob("*.d.ps")) if stdlib_dir else []
        return PsProjectConfig(
            project_name=target.stem,
            root_dir=target.parent,
            out_dir=(out_dir or (target.parent / "dist")).resolve(),
            source_files=[target],
            stub_files=stubs,
        )

    # Directory target: look for a `.psproj` first, or scan all `.ps` files
    psproj_files = list(target.glob("*.psproj"))
    if psproj_files:
        cfg = load_project(psproj_files[0], stdlib_dir=stdlib_dir)
        if out_dir is not None:
            cfg.out_dir = out_dir.resolve()
        return cfg

    stubs = sorted(stdlib_dir.rglob("*.d.ps")) if stdlib_dir else []
    for local_stub in sorted(target.rglob("*.d.ps")):
        if local_stub not in stubs:
            stubs.append(local_stub)
    sources = sorted(p for p in target.rglob("*.ps") if not p.name.endswith(".d.ps"))
    return PsProjectConfig(
        project_name=target.name,
        root_dir=target,
        out_dir=(out_dir or (target / "dist")).resolve(),
        source_files=sources,
        stub_files=stubs,
    )


def parse_and_analyze_config(
    config: PsProjectConfig,
) -> tuple[list[CompilationUnit], GlobalSymbolTable, DiagnosticEngine]:
    diag = DiagnosticEngine()
    units: list[CompilationUnit] = []

    for idx, file_path in enumerate(config.all_input_files()):
        try:
            rel_name = str(file_path.relative_to(config.root_dir))
        except ValueError:
            rel_name = file_path.name
        source_text = file_path.read_text(encoding="utf-8")
        unit = Parser.from_source(
            source_text,
            file_id=idx,
            ps_filename=rel_name,
            diagnostics=diag,
        ).parse_compilation_unit()
        units.append(unit)

    symtab = GlobalSymbolTable(default_assembly_name=config.project_name)
    if not diag.has_errors():
        analyze_compilation_units(
            units,
            symtab=symtab,
            diagnostics=diag,
            assembly_name=config.project_name,
        )

    return units, symtab, diag


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(
        prog="psc",
        description="PythonSharp (psc) — Multi-Pass Static C#-to-Python 3.11+ Transpiler",
    )
    subparsers = parser.add_subparsers(dest="command", required=True)

    # `psc check`
    check_p = subparsers.add_parser("check", help="Run Pass 1 & Pass 2 verification without emitting files")
    check_p.add_argument("target", type=Path, nargs="?", default=Path("."), help="Path to .psproj, directory, or .ps file")

    # `psc build`
    build_p = subparsers.add_parser("build", help="Compile .ps project into Python packages and line-pinned .pyc files")
    build_p.add_argument("target", type=Path, nargs="?", default=Path("."), help="Path to .psproj, directory, or .ps file")
    build_p.add_argument("-o", "--out-dir", type=Path, default=None, help="Output directory (default: dist/)")
    build_p.add_argument("--no-pyc", action="store_true", help="Disable PEP 552 .pyc bytecode emission")

    # `psc dump-ast`
    dump_p = subparsers.add_parser("dump-ast", help="Print lowered Python 3.11+ source code to stdout")
    dump_p.add_argument("target", type=Path, nargs="?", default=Path("."), help="Path to .psproj, directory, or .ps file")

    # `psc run`
    run_p = subparsers.add_parser("run", help="Compile and execute a PythonSharp entrypoint (`Class.Method` or `Main`)")
    run_p.add_argument("target", type=Path, nargs="?", default=Path("."), help="Path to .psproj, directory, or .ps file")
    run_p.add_argument("-e", "--entry", type=str, default=None, help="Entrypoint e.g. 'Program.Main'")

    args = parser.parse_args(argv)
    out_dir = getattr(args, "out_dir", None)
    config = _discover_inputs(args.target, out_dir=out_dir)

    if not config.source_files:
        print(f"ERROR: No .ps source files found in '{args.target}'.", file=sys.stderr)
        return 1

    _, symtab, diag = parse_and_analyze_config(config)
    if diag.has_errors():
        print(diag.format_all(), file=sys.stderr)
        return 1

    if args.command == "check":
        print(
            f"Check succeeded: {len(config.source_files)} .ps file(s) verified with 0 errors."
        )
        return 0

    if args.command == "dump-ast":
        planner = NamespacePackagePlanner(symtab)
        builder = PythonAstBuilder(symtab)
        import ast

        for rel_path, planned in sorted(planner.plan_modules().items()):
            mod_ast = builder.build_module(planned)
            print(f"# === Module: {rel_path} (pinned to {planned.primary_ps_filename}) ===")
            print(ast.unparse(mod_ast))
            print()
        return 0

    emit_pyc = not getattr(args, "no_pyc", False) and config.emit_pyc
    emitter = ArtifactEmitter(symtab)
    artifacts = emitter.emit_all(config.out_dir, emit_pyc=emit_pyc)

    if args.command == "build":
        print(
            f"Build succeeded: emitted {len(artifacts)} module(s) to '{config.out_dir}'."
        )
        return 0

    if args.command == "run":
        return _execute_artifacts(config.out_dir, artifacts, args.entry)

    return 0


def _execute_artifacts(
    out_dir: Path,
    artifacts: list[EmittedArtifact],
    entry_spec: str | None,
) -> int:
    sys.path.insert(0, str(out_dir))
    executed = False

    try:
        for art in artifacts:
            ns_globals: dict[str, object] = {"__name__": "__main__"}
            exec(art.code_object, ns_globals)

            if entry_spec:
                parts = entry_spec.split(".")
                cls_name = parts[0]
                method_name = parts[1] if len(parts) > 1 else "main"
                if cls_name in ns_globals:
                    cls_obj = ns_globals[cls_name]
                    fn = getattr(cls_obj, method_name, None)
                    if callable(fn):
                        fn()
                        executed = True
                        break
            else:
                # Auto-detect any class with a static `main()` method
                for val in ns_globals.values():
                    if isinstance(val, type) and hasattr(val, "main") and callable(getattr(val, "main")):
                        val.main()
                        executed = True
                        break
            if executed:
                break
    except Exception:
        traceback.print_exc()
        return 1
    finally:
        if str(out_dir) in sys.path:
            sys.path.remove(str(out_dir))

    return 0


if __name__ == "__main__":
    raise SystemExit(main())

from __future__ import annotations
from pathlib import Path
from psc.codegen import (
    ArtifactEmitter,
    EmittedArtifact,
    NamespacePackagePlanner,
    PythonAstBuilder,
)

def compile_sources(
    sources: dict[str, str],
    out_dir: Path,
    stdlib_source: str | None = None,
    assembly_name: str = "App",
    emit_pyc: bool = True,
) -> tuple[list[EmittedArtifact], DiagnosticEngine]:
    """
    High-level in-memory & disk compilation API for PythonSharp (`psc`):
      1. Parses optional `stdlib_source` (`PythonSharp.Core.d.ps`) and a mapping of
         `{ps_filename: ps_source_code}`.
      2. Runs Pass 1 (GlobalSymbolTable & SymbolCollector) and Pass 2 (SlotAndHierarchyAnalyzer,
         AccessChecker, PrimitiveResolver, OverloadResolver, LinqComprehensionFuser).
      3. Runs Pass 3 (NamespacePackagePlanner, PythonAstBuilder, ArtifactEmitter) to emit
         `.py` packages and line-pinned `.pyc` bytecode into `out_dir`.
    """
    diag = DiagnosticEngine()
    units: list[CompilationUnit] = []
    file_id = 0

    if stdlib_source is not None:
        units.append(
            Parser.from_source(
                stdlib_source,
                file_id=file_id,
                ps_filename="PythonSharp.Core.d.ps",
                diagnostics=diag,
            ).parse_compilation_unit()
        )
        file_id += 1

    for ps_filename, code in sources.items():
        units.append(
            Parser.from_source(
                code,
                file_id=file_id,
                ps_filename=ps_filename,
                diagnostics=diag,
            ).parse_compilation_unit()
        )
        file_id += 1

    if diag.has_errors():
        return [], diag

    symtab = GlobalSymbolTable(default_assembly_name=assembly_name)
    analyze_compilation_units(units, symtab=symtab, diagnostics=diag, assembly_name=assembly_name)
    if diag.has_errors():
        return [], diag

    emitter = ArtifactEmitter(symtab)
    artifacts = emitter.emit_all(out_dir=out_dir, emit_pyc=emit_pyc)
    return artifacts, diag

from __future__ import annotations
from pathlib import Path
import traceback
from typing import Any
import pytest
from psc import ArtifactEmitter, compile_sources
from psc.cli import main as cli_main


def test_pyc_bytecode_line_pinning_reports_exact_ps_file_and_line(tmp_path: Path) -> None:
    """
    Verifies Milestone 6 (Pass 3B Bytecode Line-Pinning):
      1. Compiles `Services/PaymentGateway.ps` where line 14 throws `ValueError`.
      2. Loads the emitted PEP 552 `.pyc` bytecode from disk via `ArtifactEmitter.load_code_from_pyc`.
      3. Executes `charge(-50)` and captures the native CPython traceback.
      4. Asserts that CPython's native traceback prints:
         `File "Services/PaymentGateway.ps", line 14, in charge`
         with zero custom `sys.excepthook` or third-party traceback wrappers!
    """
    ps_source = "\n".join(
        [
            "/* Line 01 */ namespace Commerce.Payments;",
            "/* Line 02 */ ",
            "/* Line 03 */ public class ValueError : Exception",
            "/* Line 04 */ {",
            "/* Line 05 */     public ValueError(string msg) { }",
            "/* Line 06 */ }",
            "/* Line 07 */ ",
            "/* Line 08 */ public class PaymentGateway",
            "/* Line 09 */ {",
            "/* Line 10 */     public int Charge(int amount)",
            "/* Line 11 */     {",
            "/* Line 12 */         if (amount <= 0)",
            "/* Line 13 */         {",
            '/* Line 14 */             throw new ValueError("Amount must be positive");',
            "/* Line 15 */         }",
            "/* Line 16 */         return amount * 100;",
            "/* Line 17 */     }",
            "/* Line 18 */ }",
        ]
    )

    artifacts, diag = compile_sources(
        sources={"Services/PaymentGateway.ps": ps_source},
        out_dir=tmp_path / "dist",
        emit_pyc=True,
    )
    assert not diag.has_errors(), diag.format_all()
    assert len(artifacts) == 1

    art = artifacts[0]
    assert art.py_path.exists()
    assert art.pyc_path is not None and art.pyc_path.exists()
    assert art.pycache_pyc_path is not None and art.pycache_pyc_path.exists()

    # Load the marshaled code object directly from the emitted `.pyc` file on disk
    loaded_code = ArtifactEmitter.load_code_from_pyc(art.pyc_path)
    assert loaded_code.co_filename == "Services/PaymentGateway.ps"

    ns_globals: dict[str, Any] = {}
    exec(loaded_code, ns_globals)

    gateway = ns_globals["PaymentGateway"]()
    assert gateway.charge(25) == 2500

    with pytest.raises(Exception) as exc_info:
        gateway.charge(-10)

    tb_lines = traceback.format_exception(
        type(exc_info.value), exc_info.value, exc_info.value.__traceback__
    )
    tb_text = "".join(tb_lines)

    # CPython's native traceback reports the exact `.ps` file and line 14!
    assert 'File "Services/PaymentGateway.ps", line 14, in charge' in tb_text


def test_cli_check_and_build_multi_file_psproj(tmp_path: Path) -> None:
    """
    Verifies `psc check` and `psc build` via `cli_main` on a multi-file `.psproj`
    project with circular cross-file references (`Order.ps` <-> `OrderCache.ps`).
    """
    proj_dir = tmp_path / "commerce_sample"
    (proj_dir / "Domain").mkdir(parents=True)
    (proj_dir / "Services").mkdir(parents=True)

    (proj_dir / "Commerce.psproj").write_text(
        """
        <Project Sdk="PythonSharp.Sdk">
          <PropertyGroup>
            <AssemblyName>Commerce.Sample</AssemblyName>
            <OutputPath>dist</OutputPath>
            <TargetPython>3.11</TargetPython>
            <EmitPyc>true</EmitPyc>
          </PropertyGroup>
        </Project>
        """.strip(),
        encoding="utf-8",
    )

    (proj_dir / "Domain" / "Order.ps").write_text(
        """
        namespace Commerce.Domain;

        using Commerce.Services;

        public record Order(int Id, decimal Total, bool IsActive);
        """.strip(),
        encoding="utf-8",
    )

    (proj_dir / "Services" / "OrderCache.ps").write_text(
        """
        namespace Commerce.Services;

        using PythonSharp.Core;
        using Commerce.Domain;

        public class OrderCache : IDisposable
        {
            private readonly Dict<int, Order> _store = new Dict<int, Order>();

            public void Put(Order order) => _store[order.Id] = order;
            public void Put(int id, decimal total) => _store[id] = new Order(id, total, true);

            public List<int> GetActiveIds()
            {
                return _store.Values()
                    .Where(o => o.IsActive)
                    .Select(o => o.Id)
                    .ToList();
            }

            public void Dispose() => _store.Clear();
        }
        """.strip(),
        encoding="utf-8",
    )

    psproj_file = proj_dir / "Commerce.psproj"

    # 1. Run `psc check`
    rc_check = cli_main(["check", str(psproj_file)])
    assert rc_check == 0

    # 2. Run `psc build`
    rc_build = cli_main(["build", str(psproj_file)])
    assert rc_build == 0

    dist_dir = proj_dir / "dist"
    assert (dist_dir / "commerce" / "__init__.py").exists()
    assert (dist_dir / "commerce" / "domain.py").exists()
    assert (dist_dir / "commerce" / "domain.pyc").exists()
    assert (dist_dir / "commerce" / "services.py").exists()
    assert (dist_dir / "commerce" / "services.pyc").exists()
