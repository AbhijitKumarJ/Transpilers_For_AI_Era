## 1. Autogeneration Script (`scripts/sync_typeshed_stdlib.py`)

This script uses `typeshed_client` to locate `.pyi` stub files (both bundled Python standard library stubs and installed third-party package stubs in `site-packages`) and feeds their ASTs into `psc.bindgen.generator.BindgenGenerator`. It supports both **1:1 exact `.pyi` replica generation** and **4-file Triple-Surface bundle generation** while maintaining SHA-256 drift hashes in `psc-bindings-manifest.json`.

```python
#!/usr/bin/env python3
"""
scripts/sync_typeshed_stdlib.py

Locates .pyi stub files across entire packages via `typeshed_client` and
generates an exact replica structured folder inside `stdlib/` using
PythonSharp's `BindgenGenerator` and `BindingManifest`.
"""

from __future__ import annotations

import argparse
import ast
import sys
from pathlib import Path
from typing import Iterable

try:
    import typeshed_client
except ImportError:
    print(
        "Error: `typeshed_client` is required for stub discovery. "
        "Run `pip install typeshed_client`.",
        file=sys.stderr,
    )
    sys.exit(1)

from psc.bindgen.generator import BindgenGenerator
from psc.bindgen.manifest import (
    BindingManifest,
    compute_content_hash,
    compute_file_hash,
)

# Modules in typeshed that are internal CPython/typing implementation details
# or already handled by PythonSharp's `stdlib/00_core/` intrinsics.
EXCLUDED_MODULES: frozenset[str] = frozenset({
    "builtins",
    "typing",
    "typing_extensions",
    "_typeshed",
    "__future__",
    "sys._monitoring",
    "ctypes.wintypes",
})


def build_search_context(python_version: tuple[int, int]) -> typeshed_client.SearchContext:
    """Create a typeshed_client SearchContext including current site-packages."""
    search_paths = [Path(p) for p in sys.path if p and Path(p).is_dir()]
    return typeshed_client.get_search_context(
        version=python_version,
        search_path=search_paths,
        platform=sys.platform,
    )


def discover_package_stubs(
    ctx: typeshed_client.SearchContext,
    target_packages: set[str] | None = None,
) -> dict[str, Path]:
    """
    Walks typeshed_client's stub index and returns a mapping of
    {dotted_module_name: Path_to_pyi_file}.
    """
    discovered: dict[str, Path] = {}

    for mod_name, pyi_path in typeshed_client.get_all_stub_files(ctx):
        if mod_name in EXCLUDED_MODULES:
            continue
        root_pkg = mod_name.split(".")[0]
        if root_pkg.startswith("_") and root_pkg not in {"__main__"}:
            continue
        if target_packages and root_pkg not in target_packages and mod_name not in target_packages:
            continue
        discovered[mod_name] = Path(pyi_path).resolve()

    return dict(sorted(discovered.items()))


def module_to_replica_dir(base_out_dir: Path, mod_name: str, pyi_path: Path) -> Path:
    """
    Computes the exact replica relative directory path for a dotted module name.
    Example:
      - `json` (json/__init__.pyi) -> base_out_dir / "json"
      - `urllib.request` (urllib/request.pyi) -> base_out_dir / "urllib" / "request"
    """
    parts = mod_name.split(".")
    return base_out_dir.joinpath(*parts)


def emit_package_replicas(
    stubs: dict[str, Path],
    out_dir: Path,
    *,
    triple_surface: bool = True,
    root_namespace: str = "Extern",
) -> tuple[int, int]:
    """
    Parses each .pyi file and emits exact replica directory structures containing
    either 4-file Triple-Surface bundles or 1:1 .d.ps files.
    """
    out_dir.mkdir(parents=True, exist_ok=True)
    manifest_path = out_dir / "psc-bindings-manifest.json"
    manifest = BindingManifest.load_from(manifest_path)

    emitted_count = 0
    skipped_count = 0

    for mod_name, pyi_path in stubs.items():
        try:
            source = pyi_path.read_text(encoding="utf-8")
            pyi_ast = ast.parse(source, filename=str(pyi_path))
        except Exception as exc:
            print(f"[WARN] Skipping {mod_name} ({pyi_path.name}): {exc}", file=sys.stderr)
            skipped_count += 1
            continue

        gen = BindgenGenerator(module_name=mod_name, root_namespace=root_namespace)
        replica_dir = module_to_replica_dir(out_dir, mod_name, pyi_path)

        if triple_surface:
            # Emits 4-file Triple-Surface bundle inside the exact module replica folder:
            # 00_{short}.primitives.d.ps, 01_{short}.raw.d.ps,
            # 02_{short}.csharp.d.ps, 03_{short}.enterprise.d.ps
            bundle = gen.generate_triple_surface_bundle(pyi_ast)
            replica_dir.mkdir(parents=True, exist_ok=True)
            surface_map: dict[str, str] = {}

            for fname, content in bundle.items():
                dest_path = replica_dir / fname
                dest_path.write_text(content, encoding="utf-8")
                emitted_count += 1
                if "primitives" in fname:
                    surface_map["primitives"] = str(dest_path.resolve())
                elif "raw" in fname:
                    surface_map["surface_1"] = str(dest_path.resolve())
                elif "csharp" in fname:
                    surface_map["surface_2"] = str(dest_path.resolve())
                elif "enterprise" in fname:
                    surface_map["surface_3"] = str(dest_path.resolve())

            manifest.add_entry(
                package_name=mod_name,
                pyi_path=pyi_path,
                pyi_hash=compute_file_hash(pyi_path),
                d_ps_path=replica_dir.resolve(),
                d_ps_hash=compute_content_hash("\n".join(bundle.values())),
                surface_files=surface_map,
            )
        else:
            # Exact 1:1 file mirror: e.g. urllib/request.pyi -> urllib/request.d.ps
            # or json/__init__.pyi -> json/__init__.d.ps
            d_ps_content = gen.generate_d_ps(pyi_ast)
            if pyi_path.stem == "__init__":
                replica_dir.mkdir(parents=True, exist_ok=True)
                dest_path = replica_dir / "__init__.d.ps"
            else:
                replica_dir.parent.mkdir(parents=True, exist_ok=True)
                dest_path = replica_dir.with_suffix(".d.ps")

            dest_path.write_text(d_ps_content, encoding="utf-8")
            emitted_count += 1

            manifest.add_entry(
                package_name=mod_name,
                pyi_path=pyi_path,
                pyi_hash=compute_file_hash(pyi_path),
                d_ps_path=dest_path.resolve(),
                d_ps_hash=compute_content_hash(d_ps_content),
            )

    manifest.write_to(manifest_path)
    return emitted_count, skipped_count


def main(argv: Iterable[str] | None = None) -> int:
    parser = argparse.ArgumentParser(
        description="Generate exact-replica .d.ps stdlib hierarchy via typeshed_client."
    )
    parser.add_argument(
        "-p",
        "--packages",
        nargs="*",
        help="Root package names to sync (e.g., json urllib asyncio numpy). Omit for full typeshed stdlib.",
    )
    parser.add_argument(
        "-o",
        "--out-dir",
        type=Path,
        default=Path("stdlib/01_python_stdlib"),
        help="Destination stdlib directory (default: stdlib/01_python_stdlib).",
    )
    parser.add_argument(
        "--single-surface",
        action="store_true",
        help="Emit 1:1 .d.ps replica files instead of 4-file Triple-Surface bundles.",
    )
    parser.add_argument(
        "--python-version",
        default="3.11",
        help="Target Python version for typeshed resolution (default: 3.11).",
    )
    args = parser.parse_args(list(argv) if argv is not None else None)

    ver_parts = tuple(int(x) for x in args.python_version.split(".")[:2])
    ctx = build_search_context((ver_parts[0], ver_parts[1]))
    target_pkgs = set(args.packages) if args.packages else None

    stubs = discover_package_stubs(ctx, target_pkgs)
    print(f"Discovered {len(stubs)} .pyi stub modules via typeshed_client.")

    emitted, skipped = emit_package_replicas(
        stubs,
        args.out_dir,
        triple_surface=not args.single_surface,
    )
    print(f"Emitted {emitted} .d.ps files ({skipped} skipped) into {args.out_dir}/")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

```

## 1. Corrected Autogeneration Script (`scripts/sync_typeshed_stdlib.py`)

This script fixes both issues:

1. It pins `REPO_ROOT / "src"` to `sys.path[0]` so the live compiler source is always used.


2. It uses `typeshed_client` to locate `.pyi` files, stages them into a temporary directory mirroring the package tree so `psc bindgen <staged_dir> -o <out_dir> [--triple-surface]` computes exact dotted module names (`json`, `urllib.request`, `asyncio.events`), invokes `psc bindgen`, and unfolds any flat dotted folders (`urllib.request/`) into an exact nested replica directory tree (`urllib/request/`) while keeping `psc-bindings-manifest.json` synchronized.



```python
#!/usr/bin/env python3
"""
scripts/sync_typeshed_stdlib.py

Discovers .pyi files via `typeshed_client`, invokes `psc bindgen <path> -o <stubs> [--triple-surface]`,
and normalizes the output into an exact nested replica folder structure under `stdlib/`.
"""

from __future__ import annotations

import argparse
import inspect
import shutil
import sys
import tempfile
from pathlib import Path
from typing import Sequence

# 1. Ensure the local repository's `src/` is loaded before any stale site-packages install
REPO_ROOT = Path(__file__).resolve().parent.parent
SRC_DIR = REPO_ROOT / "src"
if SRC_DIR.is_dir() and str(SRC_DIR) not in sys.path:
    sys.path.insert(0, str(SRC_DIR))

try:
    import typeshed_client
except ImportError:
    print(
        "ERROR: `typeshed_client` is not installed. Run: pip install typeshed_client",
        file=sys.stderr,
    )
    sys.exit(1)

from psc.bindgen import generate_bindings
from psc.bindgen.manifest import BindingManifest
from psc.cli import main as psc_cli_main

# Modules already provided by `stdlib/00_core/` or internal to CPython typechecking
EXCLUDED_MODULES: frozenset[str] = frozenset({
    "builtins",
    "typing",
    "typing_extensions",
    "_typeshed",
    "__future__",
    "sys._monitoring",
    "ctypes.wintypes",
})


def discover_typeshed_pyi_files(
    python_version: tuple[int, int],
    packages: set[str] | None = None,
) -> dict[str, Path]:
    """
    Uses `typeshed_client` to locate .pyi files across stdlib and site-packages.
    Returns {dotted_module_name: resolved_pyi_path}.
    """
    search_paths = [Path(p) for p in sys.path if p and Path(p).is_dir()]
    ctx = typeshed_client.get_search_context(
        version=python_version,
        search_path=search_paths,
        platform=sys.platform,
    )

    stubs: dict[str, Path] = {}
    for mod_name, pyi_path in typeshed_client.get_all_stub_files(ctx):
        if mod_name in EXCLUDED_MODULES:
            continue
        parts = mod_name.split(".")
        # Skip private implementation modules (e.g. `_ast`, `_collections_abc`, `os._path`)
        if any(p.startswith("_") and p not in {"__init__", "__main__"} for p in parts):
            continue
        root_pkg = parts[0]
        if packages and root_pkg not in packages and mod_name not in packages:
            continue
        stubs[mod_name] = Path(pyi_path).resolve()

    return dict(sorted(stubs.items()))


def stage_pyi_replica_tree(stubs: dict[str, Path], staging_dir: Path) -> None:
    """
    Copies discovered .pyi files into `staging_dir` preserving exact package vs. submodule
    layout (`json/__init__.pyi`, `urllib/request.pyi`) so `psc bindgen <staging_dir>`
    correctly infers dotted module names via `target.rglob("*.pyi")`.
    """
    for mod_name, src_pyi in stubs.items():
        parts = mod_name.split(".")
        if src_pyi.stem == "__init__":
            dest_pyi = staging_dir.joinpath(*parts, "__init__.pyi")
        else:
            dest_pyi = staging_dir.joinpath(*parts[:-1], f"{parts[-1]}.pyi")
        dest_pyi.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(src_pyi, dest_pyi)


def supports_triple_surface() -> bool:
    """Checks whether the active `psc.bindgen.generate_bindings` supports `triple_surface`."""
    sig = inspect.signature(generate_bindings)
    return "triple_surface" in sig.parameters


def unfold_dotted_outputs_to_replica_tree(
    raw_out_dir: Path,
    final_out_dir: Path,
    original_stubs: dict[str, Path],
    *,
    triple_surface: bool,
) -> int:
    """
    `psc bindgen` emits either:
      - Triple-Surface: `raw_out_dir / "urllib.request" / "00_request.primitives.d.ps"`
      - Single-Surface: `raw_out_dir / "urllib.request.d.ps"`
    This function moves them into the exact nested replica hierarchy:
      - Triple-Surface: `final_out_dir / "urllib" / "request" / "00_request.primitives.d.ps"`
      - Single-Surface: `final_out_dir / "urllib" / "request.d.ps"` (or `json/__init__.d.ps`)
    and rewrites `psc-bindings-manifest.json` with the real upstream `.pyi` paths and replica paths.
    """
    final_out_dir.mkdir(parents=True, exist_ok=True)
    raw_manifest_path = raw_out_dir / "psc-bindings-manifest.json"
    final_manifest_path = final_out_dir / "psc-bindings-manifest.json"

    raw_manifest = BindingManifest.load_from(raw_manifest_path)
    final_manifest = BindingManifest.load_from(final_manifest_path)

    moved_files = 0

    for mod_name, entry in raw_manifest.entries.items():
        parts = mod_name.split(".")
        orig_pyi = original_stubs.get(mod_name, Path(entry.pyi_path))

        if triple_surface:
            src_pkg_dir = raw_out_dir / mod_name
            dest_pkg_dir = final_out_dir.joinpath(*parts)
            if not src_pkg_dir.is_dir():
                continue

            dest_pkg_dir.mkdir(parents=True, exist_ok=True)
            new_surface_map: dict[str, str] = {}

            for d_ps_file in sorted(src_pkg_dir.glob("*.d.ps")):
                target_file = dest_pkg_dir / d_ps_file.name
                shutil.move(str(d_ps_file), str(target_file))
                moved_files += 1
                if "primitives" in target_file.name:
                    new_surface_map["primitives"] = str(target_file.resolve())
                elif "raw" in target_file.name:
                    new_surface_map["surface_1"] = str(target_file.resolve())
                elif "csharp" in target_file.name:
                    new_surface_map["surface_2"] = str(target_file.resolve())
                elif "enterprise" in target_file.name:
                    new_surface_map["surface_3"] = str(target_file.resolve())

            final_manifest.add_entry(
                package_name=mod_name,
                pyi_path=orig_pyi,
                pyi_hash=entry.pyi_hash,
                d_ps_path=dest_pkg_dir.resolve(),
                d_ps_hash=entry.d_ps_hash,
                surface_files=new_surface_map,
            )
        else:
            src_file = raw_out_dir / f"{mod_name}.d.ps"
            if not src_file.is_file():
                continue

            if orig_pyi.stem == "__init__":
                dest_file = final_out_dir.joinpath(*parts, "__init__.d.ps")
            else:
                dest_file = final_out_dir.joinpath(*parts[:-1], f"{parts[-1]}.d.ps")

            dest_file.parent.mkdir(parents=True, exist_ok=True)
            shutil.move(str(src_file), str(dest_file))
            moved_files += 1

            final_manifest.add_entry(
                package_name=mod_name,
                pyi_path=orig_pyi,
                pyi_hash=entry.pyi_hash,
                d_ps_path=dest_file.resolve(),
                d_ps_hash=entry.d_ps_hash,
            )

    final_manifest.write_to(final_manifest_path)
    return moved_files


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(
        description="Locate .pyi stubs via typeshed_client and run `psc bindgen` into an exact stdlib replica tree."
    )
    parser.add_argument(
        "-p",
        "--packages",
        nargs="*",
        help="Specific packages/modules to generate (e.g. json urllib asyncio). Omit for all stdlib stubs.",
    )
    parser.add_argument(
        "-o",
        "--out-dir",
        type=Path,
        default=Path("stdlib/03_typeshed_replica"),
        help="Destination replica directory (default: stdlib/03_typeshed_replica).",
    )
    parser.add_argument(
        "--root-namespace",
        default="Extern",
        help="Root namespace prefix for single-surface bindgen (default: Extern).",
    )
    parser.add_argument(
        "--triple-surface",
        action="store_true",
        help="Pass `--triple-surface` to `psc bindgen` to emit 4-file Triple-Surface bundles per module.",
    )
    parser.add_argument(
        "--python-version",
        default="3.11",
        help="Target Python version for typeshed resolution (default: 3.11).",
    )
    args = parser.parse_args(argv)

    ver_parts = tuple(int(x) for x in args.python_version.split(".")[:2])
    target_pkgs = set(args.packages) if args.packages else None

    stubs = discover_typeshed_pyi_files((ver_parts[0], ver_parts[1]), target_pkgs)
    if not stubs:
        print("No matching .pyi stubs found via typeshed_client.", file=sys.stderr)
        return 1

    print(f"Discovered {len(stubs)} .pyi module(s) via typeshed_client.")

    use_triple = args.triple_surface
    if use_triple and not supports_triple_surface():
        print(
            "WARNING: Active `psc` installation does not support `--triple-surface` "
            "(missing Milestone U3 in `psc.bindgen`). Falling back to single-surface `psc bindgen`.",
            file=sys.stderr,
        )
        use_triple = False

    with tempfile.TemporaryDirectory(prefix="psc_typeshed_") as tmp_root:
        tmp_path = Path(tmp_root)
        staging_dir = tmp_path / "pyi_staging"
        raw_out_dir = tmp_path / "bindgen_raw"

        # Step 1: Stage .pyi files in exact replica structure
        stage_pyi_replica_tree(stubs, staging_dir)

        # Step 2: Invoke `psc bindgen <staging_dir> -o <raw_out_dir> [--triple-surface]`
        cli_cmd = [
            "bindgen",
            str(staging_dir),
            "-o",
            str(raw_out_dir),
            "--root-namespace",
            args.root_namespace,
        ]
        if use_triple:
            cli_cmd.append("--triple-surface")

        print(f"Running: psc {' '.join(cli_cmd)}")
        rc = psc_cli_main(cli_cmd)
        if rc != 0:
            return rc

        # Step 3: Unfold flat dotted directories/files into exact replica hierarchy
        total_emitted = unfold_dotted_outputs_to_replica_tree(
            raw_out_dir,
            args.out_dir.resolve(),
            stubs,
            triple_surface=use_triple,
        )

    print(f"Replica generation complete: {total_emitted} .d.ps file(s) written to '{args.out_dir}'.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

## 3. Impact on the Current PythonSharp Repository

Generating `.d.ps` files across full `typeshed_client` packages impacts six areas of the existing compiler and test suite:

### 1. Namespace & Surface Mismatch vs. Curated `stdlib/01_python_stdlib/`

* **Current Curated Stubs:** Handwritten packages in `stdlib/01_python_stdlib/` use composite domain folders (`os_shutil/`, `time_datetime/`, `urllib_http/`, `pathlib_hashlib_uuid/`) and map Surface 1 directly to bare Python module namespaces (`namespace time;`, `namespace json;`), Surface 2 to `namespace PythonSharp.Time;`, and Surface 3 to domain facades (`namespace PythonSharp.System.Diagnostics;`, `namespace PythonSharp.Serialization.Json;`).


* **`psc bindgen --triple-surface` Output:** As verified in `tests/test_bindgen_triple_surface.py`, `BindgenGenerator.generate_triple_surface_bundle()` emits systematic namespaces:


* Primitives: `namespace PythonSharp.Primitives.<PascalMod>;`


* Surface 1 (Raw): `namespace PythonSharp.Raw.<PascalMod>;`


* Surface 2 (C# Bridge): `namespace PythonSharp.Bridge.<PascalMod>;`


* Surface 3 (Enterprise): `namespace PythonSharp.<PascalMod>;`




* **Impact:** Replacing `stdlib/01_python_stdlib/` or `stdlib/02_scientific_and_ai/` directly with `psc bindgen` output will break `Samples/06_triple_surface_migration/`, `Samples/07_ai_ml_triple_surface/`, and `tests/test_triple_surface.py`, which import the hand-curated namespaces and rely on custom `[InlineTemplate]` operators (such as `a.MultiplyMatrix(b)` lowering to `a @ b`). Keeping curated stubs in `01_python_stdlib/` and `02_scientific_and_ai/` while placing `typeshed_client` output in `03_typeshed_replica/` avoids breaking any existing tests.



### 2. `BindgenGenerator` AST Coverage on Real Typeshed `.pyi` Files

* **Current `BindgenGenerator` Scope:** Look at `src/psc/bindgen/generator.py`—it inspects top-level `ast.FunctionDef`, `ast.ClassDef`, and `ast.AnnAssign` nodes inside `pyi_ast.body`.


* **Typeshed Realities:**
* **Version/Platform `if` Guards:** Typeshed wraps large portions of stdlib `.pyi` files inside `if sys.version_info >= (3, 11):` or `if sys.platform != "win32":` blocks. Because `for stmt in pyi_ast.body:` in `BindgenGenerator` only checks top-level `FunctionDef`, `ClassDef`, and `AnnAssign`, any functions or classes nested inside `ast.If` version guards in Typeshed are **silently skipped** unless `ast.If` branches are flattened before generating stubs.


* **`AsyncFunctionDef` & Type Aliases:** `asyncio` and `inspect` stubs use `ast.AsyncFunctionDef`, `TypeAlias`, and PEP 695 `type X = ...` statements, which `BindgenGenerator` currently ignores if it only checks `isinstance(stmt, ast.FunctionDef)`.





### 3. Rule `PS002` Overload Collisions on Typeshed `@overload` Signatures

* PythonSharp's `OverloadResolver` enforces **Rule `PS002**`, rejecting overloaded methods whose parameter types erase to identical Python runtime types at the same arity.


* Typeshed `.pyi` files frequently define multiple `@overload` signatures differing only by `Literal["r"]` vs. `Literal["rb"]`, `TypeVar`, or generic type arguments (`list[str]` vs. `list[int]`).


* `TypeConverter` converts unknown `Literal[...]` expressions to `object` (or `string`) and `list[T]` to `List<T>` (which erases to Python `list`). When `psc` compiles a project that loads all generated stdlib stubs, `OverloadResolver` will raise `ERROR [PS002]` on those colliding signatures unless duplicate erased signatures are filtered during `BindgenGenerator` method emission.



### 4. Cross-Module Type References inside `.pyi` Packages

* In `BindgenGenerator.generate_triple_surface_bundle()`, files `01_raw`, `02_csharp`, and `03_enterprise` import `PythonSharp.Primitives.<PascalMod>`, while `00_primitives` only emits `using PythonSharp.Core;`.


* When `urllib/request.pyi` references `HTTPResponse` imported from `http.client`, `TypeConverter` leaves `HTTPResponse` as an unqualified identifier. Without emitting `using PythonSharp.Primitives.Http.Client;` (or replacing unresolved external stub types with `object`), `SymbolCollector` / `TypeChecker` will report unresolved type errors during Pass 1/Pass 2.



### 5. Compiler & LSP Startup Latency (`src/psc/project.py`)

* ` load_project()` and `find_stdlib_stubs()` in `src/psc/project.py` recursively load **every** `.d.ps` file under `stdlib/` into `all_input_files()` before parsing user `.ps` files.


* Full Python stdlib in Typeshed contains ~350 `.pyi` files ($\sim 1,400$ `.d.ps` files with `--triple-surface`). Parsing all 1,400 files on every `psc check`, `psc build`, or `psc lsp` keystroke increases Pass 1 latency from `<100ms` to several seconds unless `find_stdlib_stubs()` filters `03_typeshed_replica/` packages by the `using` / `extern using` statements present in the user's project.
